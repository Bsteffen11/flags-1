flagsList = transpose(flags)

// Rave mode mutable toggle
mutable isRave = false

// Dropdown input + Surprise Me + Rave Mode Toggle
viewof controls = {
  const select = Inputs.select(flagsList, {
    label: html`<strong>Selected Country:</strong>`,
    format: d => `${d.emoji} ${d.name}`,
    value: flagsList.find(d => d.name === "United States") || flagsList[0]
  });

  const surpriseBtn = html`<button class="btn btn-outline-primary btn-sm" style="font-weight: 600; border-radius: 6px; padding: 6px 12px; margin-left: 4px;">🎲 Surprise Me!</button>`;
  
  surpriseBtn.onclick = (e) => {
    e.preventDefault();
    const randomPick = flagsList[Math.floor(Math.random() * flagsList.length)];
    const selEl = select.querySelector("select");
    selEl.value = `${randomPick.emoji} ${randomPick.name}`;
    selEl.dispatchEvent(new Event("input", { bubbles: true }));
  };

  const raveBtn = html`<button class="btn btn-sm" style="font-weight: 700; border-radius: 6px; padding: 6px 14px; margin-left: 8px; background: linear-gradient(45deg, #ff007f, #7928ca, #00f0ff); color: white; border: none; box-shadow: 0 2px 8px rgba(255,0,127,0.4); cursor: pointer;">🪩 RAVE PARTY</button>`;
  
  raveBtn.onclick = (e) => {
    e.preventDefault();
    mutable isRave = !mutable isRave;
    const container = document.getElementById("flagAppContainer");
    if (container) {
      if (mutable isRave) {
        container.classList.add("rave-active");
        raveBtn.innerText = "🛑 STOP RAVE";
      } else {
        container.classList.remove("rave-active");
        raveBtn.innerText = "🪩 RAVE PARTY";
      }
    }
  };

  const wrapper = html`<div style="display: flex; align-items: center; gap: 10px; flex-wrap: wrap;">
    ${select}
    ${surpriseBtn}
    ${raveBtn}
  </div>`;

  Object.defineProperty(wrapper, "value", {
    get: () => select.value,
    set: (v) => { select.value = v; }
  });

  select.addEventListener("input", () => {
    wrapper.dispatchEvent(new Event("input", { bubbles: true }));
  });

  return wrapper;
}
