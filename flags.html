World Flag Similarity Explorer

flagsList = transpose(flags)

// Clean UI inputs
viewof selectedCountry = Inputs.select(
  flagsList,
  {
    label: html`<strong>Selected Country:</strong>`,
    format: d => d.name,
    value: flagsList.find(d => d.name === "United States") || flagsList[0]
  }
)

viewof viewMode = Inputs.radio(
  ["Most Similar", "Least Similar"],
  {
    label: html`<strong>Comparison:</strong>`,
    value: "Most Similar"
  }
)
comparisonFlags = {
  if (viewMode === "Most Similar") {
    return [
      { code: selectedCountry.most_similar_1_code, name: selectedCountry.most_similar_1_name, rank: "1st Most Similar" },
      { code: selectedCountry.most_similar_2_code, name: selectedCountry.most_similar_2_name, rank: "2nd Most Similar" },
      { code: selectedCountry.most_similar_3_code, name: selectedCountry.most_similar_3_name, rank: "3rd Most Similar" }
    ];
  } else {
    return [
      { code: selectedCountry.least_similar_1_code, name: selectedCountry.least_similar_1_name, rank: "1st Least Similar" },
      { code: selectedCountry.least_similar_2_code, name: selectedCountry.least_similar_2_name, rank: "2nd Least Similar" },
      { code: selectedCountry.least_similar_3_code, name: selectedCountry.least_similar_3_name, rank: "3rd Least Similar" }
    ];
  }
}

// Display target flag and the 3 comparison flags
html`
<div>
  <div class="section-title">Reference Flag</div>
  <div style="max-width: 320px;">
    <div class="flag-card">
      <span class="badge-rank target-badge">Selected Country</span>
      <h5 style="margin: 4px 0; color: #1e293b;">${selectedCountry.name}</h5>
      <div class="flag-img-container" style="height: 180px;">
        <img 
          class="flag-img" 
          src="data/${selectedCountry.code}.png" 
          alt="Flag of ${selectedCountry.name}" 
        />
      </div>
    </div>
  </div>

  <div class="section-title">Top 3 ${viewMode} Flags</div>
  <div class="results-grid">
    ${comparisonFlags.map(f => `
      <div class="flag-card">
        <span class="badge-rank">${f.rank}</span>
        <h5 style="margin: 4px 0; color: #1e293b;">${f.name}</h5>
        <div class="flag-img-container">
          <img 
            class="flag-img" 
            src="data/${f.code}.png" 
            alt="Flag of ${f.name}" 
          />
        </div>
      </div>
    `).join('')}
  </div>
</div>
`
